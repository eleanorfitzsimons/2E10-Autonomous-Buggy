void getObstacleDistance(){
  double distance;
  double duration;

  digitalWrite(US_TRIG, LOW );
  delayMicroseconds(2); 
  digitalWrite(US_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(US_TRIG, LOW);
  duration = pulseIn(US_ECHO, HIGH); // Measure pulse duration
  distance = duration / 58;  // Convert to cm
  if (distance < 35 ){
    distMode = 1;
  }
  else {
    distMode = 0;
  }
  if (distance > 13 || distance <=0){
    front_stop = 0;
  }else {
    front_stop = 1;
  }
  
// Update front stop flag
  obstacle_distance = distance;
}
