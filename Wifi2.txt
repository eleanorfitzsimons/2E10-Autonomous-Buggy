void connectClient() {
  if (!client || !client.connected()) {  // If not connected, try to accept new client
    client = server.available();
    if (client) {
      Serial.println("Client Connected");
      client.write("Hello Client");
    }
    return;  // Exit if no client is connected
  }

  while (client.available()) {  // Read all available data
    String data = client.readStringUntil('\n');
    data.trim();
    Serial.print("Received: ");
    Serial.println(data);

    // Handle GO/STOP commands
    if (data.startsWith("GO_STATUS:")) {
      if (data.indexOf("STOP") != -1) {
        stop_button = 1;
        Serial.println("STOP command received.");
        stop_buggy();
      } else  {
        stop_button = 0;
        Serial.println("GO command received.");
      }
    }

    // Handle Reference Speed Command
    else if (data.startsWith("REF_SPEED:")) {
      ref_speed = extractSpeed(data);
      Serial.print("Extracted Ref Speed: ");
      Serial.println(ref_speed);
      if (ref_speed == 0)
        front_stop = 1;
      else 
        front_stop = 0;
    }
  }
    if (millis() - lastTelemetryTime >= telemetryInterval) {
      lastTelemetryTime = millis();
      sendTelemetry();
    }

}

//parses the input data from client to extract the speed
float extractSpeed(String input) {
    input.trim(); // Remove any leading or trailing spaces/newlines

    if (!input.startsWith("REF_SPEED:")) {
        Serial.println("Failed to extract speed: REF_SPEED not found");
        return 0;
    }

    String numberStr = input.substring(10); // Extract everything after "REF_SPEED:"
    numberStr.trim(); // Ensure no leading/trailing spaces
    float speed_cmps = numberStr.toFloat();

    if (speed_cmps == 0 && numberStr != "0") { 
        Serial.println("Failed to extract speed: Conversion error");
        return 0;
    }

    Serial.println("Extracted speed: " + String(speed_cmps));
    return speed_cmps;
}
